เอกสารเชิงเทคนิค: อัลกอริทึมการทำงานของระบบ MAPS11

เอกสารฉบับนี้อธิบายชุดคำสั่งทางโปรแกรมมิ่งที่เป็นหัวใจสำคัญในการขับเคลื่อนระบบภูมิสารสนเทศเพื่อการเฝ้าระวังจุดความร้อนบริเวณสถานที่กำจัดขยะมูลฝอย

1รวบรวมข้อมูล (Data Ingestion)

ขั้นตอนนี้คือการดึงข้อมูลจุดความร้อนจากดาวเทียม Suomi NPP และ NOAA-20 ผ่านระบบ Middleware ที่ทำงานอยู่บน Google Apps Script (GAS) โดยเชื่อมต่อกับระบบ FIRMS ของ NASA ผ่าน Area API เพื่อจำกัดพื้นที่การค้นหา

โครงสร้างการเรียกข้อมูลจาก NASA FIRMS API

var nasaUrl = 'https://firms.modaps.eosdis.nasa.gov/api/area/csv/' + nasaApiKey + '/VIIRS_SNPP_NRT/' + bbox + '/2';
  • nasaApiKey: รหัสยืนยันตัวตนสำหรับสิทธิ์การเข้าถึงข้อมูลซึ่งถูกเก็บซ่อนไว้อย่างปลอดภัยใน PropertiesService ของแพลตฟอร์มเพื่อป้องกันการรั่วไหล
  • VIIRS_SNPP_NRT: แหล่งข้อมูลประเภท Near Real-Time จากเซนเซอร์ VIIRS บนดาวเทียม Suomi NPP ที่มีจุดเด่นเรื่องการอัปเดตข้อมูลที่รวดเร็วทันต่อเหตุการณ์
  • bbox (Bounding Box): ชุดพิกัดขอบเขตพื้นที่ที่ต้องการดึงข้อมูลโดยระบบกำหนดค่าไว้ที่ 100.0 13.0 105.0 17.5 ซึ่งเป็นการสร้างกรอบสี่เหลี่ยมคลุมพื้นที่ภาคตะวันออกเฉียงเหนือตอนล่างเพื่อลดปริมาณข้อมูลที่ต้องดาวน์โหลด
  • พารามิเตอร์เวลา (/2): กำหนดให้ NASA ส่งข้อมูลจุดความร้อนที่ตรวจพบในช่วง 2 วันที่ผ่านมาทั้งหมดกลับมาเพื่อนำมาประมวลผลต่อ

  • แปลงไฟล์และสกัดข้อมูล: ข้อมูลที่ได้มาจะอยู่ในรูปแบบข้อความ CSV ระบบจึงแยกบรรทัดและคอลัมน์เพื่อสกัดเฉพาะค่าที่สำคัญได้แก่ละติจูด ลองจิจูด วันที่ เวลา และระดับความเชื่อมั่น

  • คัดกรองตามขอบเขตรับผิดชอบ: นำพิกัดไปตรวจสอบกับรูปทรง GeoJSON เพื่อยืนยันว่าจุดนั้นตกลงในพื้นที่ 4 จังหวัดเป้าหมายอย่างแท้จริงก่อนบันทึกลงฐานข้อมูล

2วิเคราะห์เชิงพื้นที่ (Spatial Analysis & Filtering)

เมื่อได้ข้อมูลจุดความร้อนระบบจะใช้ Turf.js สร้างพิกัดและตรวจสอบเชิงเรขาคณิตว่าจุดนั้นตกลงในรูปหลายเหลี่ยม (Polygon) ซึ่งเป็นขอบเขต 4 จังหวัด (นครราชสีมา ชัยภูมิ บุรีรัมย์ สุรินทร์) หรือไม่

let turfPt = turf.point([lon, lat]); // แปลงพิกัดเป็น Turf Point
let strictProvName = null;

if (boundaryGeoJSON && boundaryGeoJSON.features) {
    for (let b of boundaryGeoJSON.features) {
        // เช็คก่อนว่าพิกัดอยู่ในกรอบสี่เหลี่ยม (BBox) ของจังหวัดนั้นไหม เพื่อลดภาระการประมวลผล
        if (lon >= b.bbox[0] && lat >= b.bbox[1] && lon <= b.bbox[2] && lat <= b.bbox[3]) {
            // เช็คความแม่นยำระดับเส้นขอบเขตจังหวัดว่าพิกัดตกอยู่ข้างในจริงๆ หรือไม่
            if (turf.booleanPointInPolygon(turfPt, b)) {
                let propStr = JSON.stringify(b.properties);
                for (let target of targetProvs) {
                    if (propStr.includes(target)) {
                        strictProvName = target; // พบว่าอยู่ใน 4 จังหวัดเป้าหมาย
                        break;
                    }
                }
            }
        }
        if (strictProvName) break;
    }
}

3ประเมินระดับความเสี่ยง (Risk Assessment)

นำพิกัดสถานที่กำจัดขยะมูลฝอยมาคำนวณระยะทางเทียบกับจุดความร้อนโดยใช้โมดูล distance ของ Turf.js หากพบว่าอยู่ห่างกันไม่เกิน 0.5 กิโลเมตรจะถือเป็นพื้นที่เสี่ยงไฟไหม้บ่อขยะ

const RISK_RADIUS_KM = 0.5; // รัศมี 500 เมตร

function calculateWasteFireRisk() {
    let risksFound = [];
    
    // วนลูปบ่อขยะทุกแห่ง
    landfillMarkers.forEach(site => {
        let sitePt = turf.point([site.marker.getLatLng().lng, site.marker.getLatLng().lat]);
        let nearHotspots = 0;

        // วนลูปจุดความร้อนที่ผ่านการกรองแล้ว
        currentFilteredHotspots.forEach(f => {
            let hotPt = turf.point([f.geometry.coordinates[0], f.geometry.coordinates[1]]);
            // คำนวณระยะห่างระหว่างบ่อขยะกับจุดความร้อนเป็นกิโลเมตร
            let distance = turf.distance(sitePt, hotPt, {units: 'kilometers'});
            
            // ถ้าระยะห่างอยู่ในเกณฑ์ความเสี่ยงให้นับเพิ่ม
            if (distance <= RISK_RADIUS_KM) {
                nearHotspots++;
            }
        });

        if (nearHotspots > 0) {
            risksFound.push({
                siteName: site.name,
                latlng: site.marker.getLatLng(),
                count: nearHotspots
            });
        }
    });
    displayFireRiskAlerts(risksFound);
}

4วิเคราะห์การแพร่กระจาย (Plume Dispersion & Downwind Impact)

ดึงข้อมูลพารามิเตอร์ด้านลมจาก Open-Meteo API มาสร้างรูปร่างควันไฟ (Plume) ไปยังพื้นที่ท้ายลมและแปลงพิกัดกลับเป็นที่อยู่โดยใช้ Longdo Map API เพื่อระบุชุมชนที่ได้รับผลกระทบ

// 1. ดึงข้อมูลสภาพอากาศและลม ณ พิกัดเกิดเหตุ
fetch(`https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current=wind_speed_10m,wind_direction_10m`)
    .then(res => res.json())
    .then(data => {
        let speed = data.current.wind_speed_10m;
        let dir = data.current.wind_direction_10m;
        
        // กลับทิศทางลมเพื่อหาทิศที่ควันจะลอยไป (พื้นที่ท้ายลม)
        const smokeAngle = (dir + 180) % 360;
        const bearing = smokeAngle > 180 ? smokeAngle - 360 : smokeAngle; 

        // 2. ใช้ Turf.js สร้างพิกัดเป้าหมายระยะ 2.5 กม และ 5 กม ตามทิศทางลม

        const pt2 = turf.destination([lon, lat], 5, bearing, {units: 'kilometers'}).geometry.coordinates;

        // 3. นำพิกัดปลายทางไปแปลงเป็นที่อยู่ด้วย Longdo API (Reverse Geocoding)
        Promise.all([
            fetch(`https://api.longdo.com/map/services/address?lon=${pt1[0]}&lat=${pt1[1]}&key=${longdoApiKey}`).then(r => r.json()),
            fetch(`https://api.longdo.com/map/services/address?lon=${pt2[0]}&lat=${pt2[1]}&key=${longdoApiKey}`).then(r => r.json())
        ]).then(results => {

5แจ้งเตือนและตอบสนอง (Alerting System)

สร้างองค์ประกอบส่วนต่อประสานกราฟิกเพื่อเร่งรัดกระบวนการตอบสนองซึ่งรวมถึงการวาดวงกลมแจ้งเตือนความเสี่ยงบนแผนที่และระบบ Toast Notification

function displayFireRiskAlerts(risks) {
    if (risks.length > 0) {
        risks.forEach(risk => {
            // วาดวงกลมสีแดงกระพริบทับพิกัดบ่อขยะเพื่อเน้นย้ำความสำคัญ
            L.circle(risk.latlng, {
                radius: RISK_RADIUS_KM * 1000, // แปลงรัศมีเป็นหน่วยเมตร
                color: '#d50000', 
                fillColor: '#ff1744', 
                fillOpacity: 0.3, 
                weight: 3,
                className: 'blink-alert' // เรียกใช้คลาส CSS ที่มีแอนิเมชันกระพริบ
            }).addTo(riskAlertLayer)
              .bindPopup(`<div style="text-align:center;"><b>🚨 พื้นที่เสี่ยงสูง!</b><br>บ่อขยะ: <span style="color:#b91c1c; font-weight:bold;">${risk.siteName}</span><br>พบจุดความร้อน ${risk.count} จุด ในรัศมี ${RISK_RADIUS_KM} กม.</div>`);
        });
        
        // แจ้งเตือนมุมขวาบนหน้าจอเพื่อให้เจ้าหน้าที่สังเกตเห็นทันที
        showToast(`⚠️ เฝ้าระวังฉุกเฉิน พบจุดความร้อนใกล้บ่อขยะ ${risks.length} แห่ง`, "error");
    }
}

6การแสดงผลแผนที่และการโต้ตอบ (Map Rendering & Interactivity)

ฟังก์ชันกราฟิกต่างๆ ของโปรแกรมถูกสร้างขึ้นบนโครงสร้างของไลบรารี Leaflet.js ทั้งสิ้นโดยมีรายละเอียดการประยุกต์ใช้งานดังนี้

  • การจัดผังและซ้อนทับเลเยอร์: โค้ดมีการเรียกใช้คำสั่งควบคุมเลเยอร์เพื่อสร้างเมนูที่มุมขวาบนช่วยให้เจ้าหน้าที่สลับดูแผนที่ฐานแบบต่างๆ ได้อย่างอิสระทั้งแผนที่ถนน OpenStreetMap ภาพถ่ายดาวเทียม Esri และ Google Maps แบบไฮบริดรวมถึงสามารถเลือกเปิดปิดชั้นข้อมูลซ้อนทับต่างๆ เช่น จุดความร้อน NASA สถานที่กำจัดขยะ พื้นที่แจ้งเตือน และจุดเกิดเหตุด่วนได้ตามความต้องการใช้งานเฉพาะหน้า
  • การจัดการจุดความร้อนจำนวนมาก: ระบบมีการเรียกใช้ปลั๊กอิน Leaflet.markercluster เพื่อรวบกลุ่มหมุดจุดความร้อนจากดาวเทียม NASA เข้าด้วยกันแบบอัตโนมัติเมื่อซูมแผนที่ออกและจะกระจายตัวออกเป็นรูปแบบใยแมงมุมเมื่อซูมเข้าใกล้พื้นที่เป้าหมายเทคนิคนี้ช่วยลดภาระการประมวลผลทำให้หน้าเว็บเบราว์เซอร์ไม่หน่วง
  • การวาดรัศมีและพื้นที่แจ้งเตือนภัย: โค้ดใช้คำสั่งวาดรูปทรงเรขาคณิตซ้อนทับลงบนแผนที่โดยตรงเช่นการใช้วงกลมสีแดงแบบกระพริบในรัศมี 500 เมตรรอบสถานที่กำจัดขยะมูลฝอยรวมถึงการวาดรูปทรงหลายเหลี่ยมแบบโปร่งแสงเพื่อจำลองพื้นที่รัศมี 5 กิโลเมตรที่ควันไฟอาจแพร่กระจายไปถึง
  • การโต้ตอบผ่านหน้าต่างข้อมูล: เมื่อมีการคลิกที่หมุดจุดความร้อนระบบจะสั่งให้ป๊อปอัปเด้งขึ้นมาแสดงผลลัพธ์จากการคำนวณเบื้องหลังทันทีซึ่งครอบคลุมทั้งข้อมูลพิกัดละติจูดลองจิจูดระดับตำบลความเร็วลมและข้อความแจ้งเตือนพร้อมปุ่มเปิดระบบนำทางไปยังจุดเกิดเหตุผ่าน Google Maps

7ระบบแจ้งเตือนอัจฉริยะ (Toast Notification System)

เพื่อประสบการณ์ใช้งานที่ดี (UX) ระบบ MAPS11 ได้พัฒนากลไกแจ้งเตือนแบบ Non-blocking UI ที่จะแสดงข้อความสถานะต่างๆ ลอยขึ้นมาที่มุมขวาบนของหน้าจอโดยไม่ขัดจังหวะการทำงานของเจ้าหน้าที่และจะจางหายไปเองโดยอัตโนมัติ

  • การแบ่งระดับความสำคัญ (Severity Levels): ระบบรองรับการแสดงผล 4 ระดับ ได้แก่ Success (สีเขียว) สำเร็จ Error (สีแดง) ผิดพลาด/ฉุกเฉิน Warning (สีเหลือง) เฝ้าระวัง และ Info (สีฟ้า) แจ้งสถานะทั่วไป
  • การจัดการหน่วยความจำ (DOM Management): ทุกครั้งที่มีการแจ้งเตือนฟังก์ชันจะสร้าง DOM Element ใหม่ขึ้นมาและใช้ setTimeout ลบองค์ประกอบนั้นทิ้งไปหลังจากผ่านไป 3.3 วินาทีเพื่อป้องกันไม่ให้หน้าเว็บสะสมข้อมูลขยะ
  • การแสดงผลซ้อนทับ (Stacking): โครงสร้าง CSS แบบ Flexbox ใน #toast-container ช่วยให้ระบบสามารถแสดงข้อความแจ้งเตือนหลายรายการพร้อมกันได้โดยจะเรียงต่อกันลงมาอย่างเป็นระเบียบ
// ฟังก์ชันหลักสำหรับเรียกใช้ Toast Notification
function showToast(message, type = 'info') {
    const container = document.getElementById('toast-container');
    if (!container) return;
    
    // สร้างกล่องข้อความใหม่และใส่คลาสตามประเภท
    const toast = document.createElement('div');
    toast.className = `toast ${type}`;
    
    // กำหนดไอคอนตามประเภทการแจ้งเตือน
    let icon = 'ℹ️';
    if (type === 'success') icon = '✅';
    if (type === 'error') icon = '❌';
    if (type === 'warning') icon = '⚠️';
    
    toast.innerHTML = `<span style="margin-right:8px; font-size:16px;">${icon}</span> ${message}`;
    container.appendChild(toast);
    
    // ตั้งเวลาให้ข้อความจางหายและลบตัวเองออกไป
    setTimeout(() => {

8การวิเคราะห์ข้อมูลเชิงพื้นที่ (Spatial Analysis) ด้วย Turf.js

ระบบเรียกใช้ไลบรารี Turf.js เป็นเครื่องมือหลักในการวิเคราะห์และคำนวณข้อมูลเชิงพื้นที่ทางภูมิศาสตร์โดยมีขอบเขตการทำงานดังนี้

  • การสร้างพิกัดและกล่องขอบเขต (Point & Bounding Box): ใช้ turf.point แปลงข้อมูลพิกัดละติจูดและลองจิจูดให้เป็นจุดอ้างอิงเชิงพื้นที่และใช้ turf.bbox สร้างกล่องกรอบสี่เหลี่ยมคลุมพื้นที่เพื่อนำไปใช้คัดกรองข้อมูลจุดความร้อนในชุดแรก
  • การตรวจสอบจุดตกกระทบในพื้นที่ (Point in Polygon): ใช้คำสั่ง turf.booleanPointInPolygon เพื่อตรวจสอบอย่างละเอียดว่าจุดความร้อนตกอยู่ภายในพื้นที่รูปหลายเหลี่ยมของ 4 จังหวัดเป้าหมายหรือไม่
  • การคำนวณระยะทางเพื่อหาระยะอันตราย (Distance Calculation): ใช้คำสั่ง turf.distance วัดระยะห่างระหว่างจุดพิกัดสถานที่กำจัดขยะกับจุดความร้อนเพื่อทริกเกอร์การแจ้งเตือนพื้นที่เสี่ยงภัยฉุกเฉิน
  • การคำนวณพิกัดท้ายลมเพื่อวาดแนวควันไฟ (Destination & Plume Generation): ใช้คำสั่ง turf.destination ยิงพิกัดจำลองออกไปจากจุดเกิดเหตุตามองศาของทิศทางลมเพื่อหาพิกัดปลายทางที่ระยะ 2.5 กิโลเมตรและ 5 กิโลเมตรเพื่อระบุชื่อชุมชนท้ายลมรวมถึงสร้างจุดขอบซ้ายและขวาประกอบร่างเป็นรูปทรงกรวยจำลองแนวควัน
// 1. การสร้างจุดและ Bounding Box
let turfPt = turf.point([lon, lat]);
boundaryGeoJSON.features.forEach(f => f.bbox = turf.bbox(f));

// 2. การตรวจสอบจุดตกกระทบในพื้นที่ Polygon
let isInside = turf.booleanPointInPolygon(turfPt, b);

// 3. การคำนวณระยะทางระหว่างจุด (หาพิกัดใกล้เคียงบ่อขยะ)
let distance = turf.distance(sitePt, hotPt, {units: 'kilometers'});

// 4. การหาพิกัดปลายทางจากทิศทางลมและระยะทางเพื่อสร้างแนวควันไฟ
const leftPt = turf.destination([lon, lat], distance, leftBearing, {units: 'kilometers'}).geometry.coordinates;
const rightPt = turf.destination([lon, lat], distance, rightBearing, {units: 'kilometers'}).geometry.coordinates;

10ระบบแคชและฐานข้อมูลชั่วคราว (Data Caching & Storage)

เพื่อลดความหน่วงในการโหลดข้อมูลจาก API ภายนอกและป้องกันข้อจำกัดด้านการเรียกใช้งานระบบจะใช้ Google Sheets เป็นเสมือนฐานข้อมูลชั่วคราวโดยเก็บข้อมูลจุดความร้อนที่ผ่านการคัดกรองแล้วไว้ในชีต Hotspot_Map และเก็บสถิติคุณภาพอากาศไว้ในชีตข้อมูลรายวัน

function getCachedHotspots() {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOTSPOT_SHEET);
    var counts = { "นครราชสีมา": 0, "ชัยภูมิ": 0, "บุรีรัมย์": 0, "สุรินทร์": 0 };
    var points = [];
    
    // อ่านข้อมูลจากแคชเพื่อส่งให้หน้าเว็บทันที
    if (sheet) {
        var data = sheet.getDataRange().getValues();
        for (var i = 1; i < data.length; i++) {
            var prov = data[i][2];
            if (prov && counts[prov] !== undefined) {
                counts[prov]++;
                points.push({ lat: data[i][0], lon: data[i][1], prov: prov /* ข้อมูลอื่นๆ */ });
            }
        }
    }
    return ContentService.createTextOutput(JSON.stringify({ counts: counts, points: points })).setMimeType(ContentService.MimeType.JSON);
}

11อัลกอริทึมคัดกรองขอบเขตระดับเซิร์ฟเวอร์ (Server-Side Ray-Casting)

นอกจากตรวจจับบนหน้าเว็บแล้วระบบหลังบ้านยังมีอัลกอริทึมสำหรับตรวจพิกัด (Point in Polygon) เพื่อกรองเฉพาะจุดความร้อนที่อยู่ในขอบเขตพื้นที่รับผิดชอบก่อนบันทึกลงฐานข้อมูลช่วยลดปริมาณข้อมูลขยะและเพิ่มความเร็วในการตอบสนอง

function isPointInPolygon(pt, ring) {
    var inside = false;
    for (var i = 0, j = ring.length - 1; i < ring.length; j = i++) {
        if (((ring[i][1] > pt[1]) != (ring[j][1] > pt[1])) && 
            (pt[0] < (ring[j][0] - ring[i][0]) * (pt[1] - ring[i][1]) / (ring[j][1] - ring[i][1]) + ring[i][0])) {
            inside = !inside;
        }
    }
    return inside;
}

12การประมวลผลอัตโนมัติตามรอบเวลา (Time-Driven Triggers)

ระบบกำหนดเวลาจะสั่งรันสคริปต์แบบอัตโนมัติตามช่วงเวลาที่กำหนดเช่นการดึงข้อมูล NASA ล่วงหน้าในรอบเช้าและเย็นและการสรุปยอดเพื่อบันทึกลงรายงานสถิติประจำวัน

function setupDailyTriggers() {
    // ล้างรอบเวลาเดิมเพื่อป้องกันการทำงานซ้ำซ้อน
    var triggers = ScriptApp.getProjectTriggers();
    triggers.forEach(t => ScriptApp.deleteTrigger(t));

    // 1. ดึงข้อมูล NASA รอบเช้า (07:15) และรอบเย็น (17:15)
    [7, 17].forEach(h => {
        ScriptApp.newTrigger('fetchAndCacheNasaHotspots').timeBased().atHour(h).nearMinute(15).everyDays(1).create();
    });

    // 2. อัปเดตสถิติ PM2.5 และจุดความร้อนรายวัน (07:45)
    ScriptApp.newTrigger('updateDailyPM25').timeBased().atHour(7).nearMinute(45).everyDays(1).create();
}
